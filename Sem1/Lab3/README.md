# 1. ЧАСТЬ 1. КЛАССЫ И ПЕРЕГРУЗКА ОПЕРАЦИЙ В C++

## 1.1. Цель работы

Изучение классов и объектов языка C++, освоение конструкторов, деструктора и перегрузки стандартных операций. Реализация класса для работы с векторами, динамическое выделение памяти и демонстрация копирования и перемещения объектов.

## 1.2. Условие задачи

Создать класс `Vector`, задающий вектор размерности `n`. Элементы вектора имеют тип `double` и хранятся в динамическом массиве. Класс должен содержать конструктор без параметров для создания пустого вектора, конструктор для создания вектора на основе обычного массива, конструктор копирования, конструктор перемещения и деструктор.

Для варианта № 11 необходимо перегрузить операцию `*` для вычисления скалярного произведения двух векторов одинаковой размерности. Результатом операции должно быть одно число типа `double`.

Также в соответствии с общим условием части 1 перегружаются операции `[]` для доступа к элементам вектора, `=` для копирующего и перемещающего присваивания, `<<` для вывода и `>>` для ввода вектора. Входные данные считываются из файла `input.txt`, результаты записываются в файл `output.txt`.

### Файл `Part1_Vector/vector.h`

```cpp
#ifndef VECTOR_H
#define VECTOR_H

#include <iostream>

using namespace std;

// Класс для работы с вектором
class Vector {
private:
    double* p; // Указатель на динамический массив
    int n;     // Размер вектора

public:
    Vector(); // Создание пустого вектора
    Vector(const double* a, int n); // Создание из обычного массива
    explicit Vector(int n); // Создание вектора заданного размера
    Vector(const Vector& v); // Конструктор копирования
    Vector(Vector&& v) noexcept; // Конструктор перемещения
    ~Vector(); // Освобождение памяти

    int size() const; // Получение размера вектора

    double& operator[](int i); // Доступ к элементу
    const double& operator[](int i) const;

    Vector& operator=(const Vector& v); // Копирующее присваивание
    Vector& operator=(Vector&& v) noexcept; // Перемещающее присваивание

    double operator*(const Vector& v) const; // Скалярное произведение
};

// Перегрузка операций ввода и вывода
ostream& operator<<(ostream& os, const Vector& v);
istream& operator>>(istream& is, Vector& v);

#endif
```

### Файл `Part1_Vector/vector.cpp`

```cpp
#include "vector.h"

#include <stdexcept>
#include <utility>

using namespace std;

// Создаем пустой вектор
Vector::Vector() : p(nullptr), n(0) {}

// Создаем вектор на основе обычного массива
Vector::Vector(const double* a, int sz) : p(nullptr), n(0) {
    if (sz < 0 || (sz > 0 && a == nullptr)) {
        throw invalid_argument("Неверный массив или размер");
    }

    if (sz > 0) {
        p = new double[sz];
        n = sz;

        for (int i = 0; i < n; i++) {
            p[i] = a[i];
        }
    }
}

// Создаем вектор заданного размера
Vector::Vector(int sz) : p(nullptr), n(0) {
    if (sz < 0) {
        throw invalid_argument("Размер не может быть отрицательным");
    }

    if (sz > 0) {
        p = new double[sz]{};
        n = sz;
    }
}

// Создаем независимую копию вектора
Vector::Vector(const Vector& v) : p(nullptr), n(v.n) {
    if (n > 0) {
        p = new double[n];

        for (int i = 0; i < n; i++) {
            p[i] = v.p[i];
        }
    }
}

// Перемещаем данные, не копируя элементы
Vector::Vector(Vector&& v) noexcept : p(v.p), n(v.n) {
    v.p = nullptr;
    v.n = 0;
}

// Освобождаем память
Vector::~Vector() {
    delete[] p;
}

// Возвращаем размер вектора
int Vector::size() const {
    return n;
}

// Возвращаем элемент по индексу
double& Vector::operator[](int i) {
    if (i < 0 || i >= n) {
        throw out_of_range("Неверный индекс элемента");
    }

    return p[i];
}

// Доступ к элементу без изменения
const double& Vector::operator[](int i) const {
    if (i < 0 || i >= n) {
        throw out_of_range("Неверный индекс элемента");
    }

    return p[i];
}

// Копирующее присваивание
Vector& Vector::operator=(const Vector& v) {
    if (this != &v) {
        double* q = nullptr;

        if (v.n > 0) {
            q = new double[v.n];

            for (int i = 0; i < v.n; i++) {
                q[i] = v.p[i];
            }
        }

        delete[] p;
        p = q;
        n = v.n;
    }

    return *this;
}

// Перемещающее присваивание
Vector& Vector::operator=(Vector&& v) noexcept {
    if (this != &v) {
        delete[] p;
        p = v.p;
        n = v.n;
        v.p = nullptr;
        v.n = 0;
    }

    return *this;
}

// Вычисляем скалярное произведение двух векторов
double Vector::operator*(const Vector& v) const {
    if (n != v.n) {
        throw invalid_argument("Размеры векторов должны совпадать");
    }

    double sum = 0;

    for (int i = 0; i < n; i++) {
        sum += p[i] * v.p[i];
    }

    return sum;
}

// Выводим вектор в поток
ostream& operator<<(ostream& os, const Vector& v) {
    os << "Размер вектора: " << v.size() << endl;
    os << "Элементы: ";

    for (int i = 0; i < v.size(); i++) {
        os << v[i] << " ";
    }

    os << endl;
    return os;
}

// Считываем размер и элементы вектора из потока
istream& operator>>(istream& is, Vector& v) {
    int sz;

    if (!(is >> sz)) {
        return is;
    }

    if (sz < 0) {
        is.setstate(ios::failbit);
        return is;
    }

    Vector tmp(sz);

    for (int i = 0; i < sz; i++) {
        if (!(is >> tmp[i])) {
            return is;
        }
    }

    // Передаем данные считанного вектора
    v = move(tmp);
    return is;
}
```

```cpp
#include <iostream>
#include <fstream>
#include <utility>

#include "vector.h"

int main() {

    ifstream in("input.txt");

        cout << "Ошибка открытия файлов!" << endl;
        return 1;

    Vector a, b;

    // Считываем два вектора из файла

        out << "Ошибка чтения данных из файла!" << endl;
        cout << "Ошибка чтения данных из файла!" << endl;
        return 1;
    }

    out << "Первый вектор:" << endl;
    out << a << endl;

    out << "Второй вектор:" << endl;

    // Проверяем совпадение размеров
    if (a.size() != b.size()) {
        out << "Размеры векторов не совпадают!" << endl;
        cout << "Размеры векторов не совпадают!" << endl;
        return 1;
    }

    // Вычисляем скалярное произведение
    out << "Скалярное произведение векторов: "

    // Проверяем обращение к элементу по индексу
```

# 2. ЧАСТЬ 2. НАСЛЕДОВАНИЕ КЛАССОВ В C++

## 2.1. Цель работы

Изучение наследования классов в языке C++, создание базового и производного классов, переопределение методов и демонстрация статического и динамического связывания.

## 2.2. Условие задачи

Создать базовый класс `School` («Школа»). Класс содержит название школы и количество обучающихся, конструктор для инициализации полей и функцию для печати параметров школы.

Создать производный класс `SpecSchool` («Специализированная школа»), который дополнительно содержит название специализации. В конструкторе производного класса должен вызываться конструктор базового класса. Функция печати должна быть переопределена, причём сначала вызывается функция базового класса, а затем выводится специализация.

Создать по одному объекту каждого класса, продемонстрировать работу методов, статическое и динамическое связывание. Создать вектор указателей на объекты базового и производного классов и вывести данные всех объектов.

## 2.3. Текст программы с комментариями

### Файл `Part2_Inheritance/main.cpp`

```cpp
#include <iostream>
#include <string>
#include <vector>

using namespace std;

// Базовый класс "Школа"
class School {
protected:
    string name; // Название школы
    int cnt;     // Количество учеников

public:
    // Конструктор для заполнения полей
    School(const string& n, int c) {
        name = n;
        cnt = c;
    }

    // Виртуальная функция для печати данных
    virtual void print() const {
        cout << "Название школы: " << name << endl;
        cout << "Количество учеников: " << cnt << endl;
    }

    // Обычная функция для демонстрации статического связывания
    void printStatic() const {
        cout << "Название школы: " << name << endl;
        cout << "Количество учеников: " << cnt << endl;
    }

    // Виртуальный деструктор базового класса
    virtual ~School() = default;
};

// Производный класс "Специализированная школа"
class SpecSchool : public School {
private:
    string spec; // Направление специализации

public:
    // Сначала вызывается конструктор базового класса
    SpecSchool(const string& n, int c, const string& s)
        : School(n, c), spec(s) {}

    // Переопределяем функцию печати
    void print() const override {
        School::print(); // Выводим основные данные школы
        cout << "Специализация: " << spec << endl;
    }

    // Функция для демонстрации статического связывания
    void printStatic() const {
        School::printStatic();
        cout << "Специализация: " << spec << endl;
    }
};

int main() {
    // Создаем по одному объекту каждого класса
    School s("Школа № 12", 650);
    SpecSchool sp("Лицей № 1", 420, "Физика и математика");

    cout << "=== ВЫЗОВ ФУНКЦИЙ ОБЪЕКТОВ ===" << endl;

    cout << endl << "Обычная школа:" << endl;
    s.print();

    cout << endl << "Специализированная школа:" << endl;
    sp.print();

    // Создаем указатель базового класса
    School* p = &sp;

    cout << endl << "=== СТАТИЧЕСКОЕ СВЯЗЫВАНИЕ ===" << endl;

    cout << "Вызов функции объекта специализированной школы:" << endl;
    sp.printStatic();

    cout << endl << "Вызов через указатель базового класса:" << endl;
    p->printStatic();

    // Здесь вызывается функция базового класса,
    // потому что printStatic не объявлена виртуальной.

    cout << endl << "=== ДИНАМИЧЕСКИЙ ПОЛИМОРФИЗМ ===" << endl;

    // Вызывается функция реального типа объекта
    p->print();

    // Создаем вектор указателей на объекты обоих классов
    vector<School*> v = {&s, &sp};

    cout << endl << "=== СПИСОК ВСЕХ ШКОЛ ===" << endl;

    for (School* q : v) {
        q->print();
        cout << endl;
    }

    return 0;
}
```
