# 1. ЧАСТЬ 1. ИЗУЧЕНИЕ ОПЕРАТОРОВ ЯЗЫКА C++

## 1.1. Цель работы

Изучение циклических алгоритмов на языке C++, освоение методов численного интегрирования и применение циклов для вычисления определённого интеграла с заданной точностью.

## 1.2. Условие задачи

Вычислить определённый интеграл методом трапеций с заданной точностью ε = 10⁻², 10⁻⁴, 10⁻⁶. Исследовать зависимость количества разбиений от заданной точности вычислений.

Подынтегральная функция:

f(x) = eˣ · (1 + sin(x)) / (1 + cos(x))

Пределы интегрирования:

a = 0, b = 2.

Для вычисления используется метод трапеций с последовательным удвоением количества разбиений. Оценка погрешности определяется по формуле:

Δ = |I₂ₙ − Iₙ| / 3

где Iₙ — приближённое значение интеграла при n разбиениях.

Вычисления продолжаются до тех пор, пока оценка погрешности не станет меньше заданной точности.

## 1.3. Текст программы

```cpp
#include <iostream>
#include <cmath>
#include <iomanip>
#include <windows.h>

using namespace std;

// Подынтегральная функция
double f(double x) {
    return exp(x) * (1 + sin(x)) / (1 + cos(x));
}

// Вычисление интеграла методом трапеций
double trap(double a, double b, int n) {
    double h = (b - a) / n;
    double sum = (f(a) + f(b)) / 2.0;

    for (int i = 1; i < n; i++) {
        double x = a + i * h;
        sum += f(x);
    }

    return sum * h;
}

// Вычисление с заданной точностью
double Integ(double eps) {
    const double a = 0.0;
    const double b = 2.0;
    const int max_n = 16777216;

    int n = 1;
    double previous = trap(a, b, n);

    while (n < max_n) {
        n *= 2;

        double current = trap(a, b, n);
        double error = abs(current - previous) / 3.0;

        if (error < eps) {
            return current;
        }

        previous = current;
    }

    return previous;
}

int main() {
    SetConsoleOutputCP(CP_UTF8);
    cout << fixed << setprecision(10);

    double epsilons[] = {0.01, 0.0001, 0.000001};

    for (double eps : epsilons) {
        double result = Integ(eps);

        cout << "Точность: " << eps << endl;
        cout << "Значение интеграла: " << result << endl;
    }

    return 0;
}
```

# 2. ЧАСТЬ 2. ИЗУЧЕНИЕ МАССИВОВ И СТРУКТУР ЯЗЫКА C++

## 2.1. Цель работы

Изучение пользовательских структур языка C++, освоение работы с динамическими контейнерами и реализация алгоритмов обработки данных студентов и их оценок по различным дисциплинам.

## 2.2. Условие задачи

Разработать программу для хранения и обработки данных студентов.

Для каждого студента необходимо хранить:
- ФИО;
- список изучаемых дисциплин;
- оценку по каждой дисциплине.

Программа должна выполнять следующие действия:

1. Запрашивать количество студентов.
2. Запрашивать количество дисциплин, при этом число дисциплин должно быть не меньше четырёх.
3. Считывать названия дисциплин.
4. Считывать ФИО студентов и оценки по каждой дисциплине.
5. Проверять, что каждая оценка находится в диапазоне от 2 до 5 включительно.
6. Выводить информацию обо всех студентах и их средние баллы.
7. Определять наибольший средний балл и выводить студентов, которые его имеют.

Для хранения информации используются структуры `Discipline` и `Student`, а также динамический контейнер `vector`.

## 2.3. Текст программы

```cpp
#include <iostream>
#include <iomanip>
#include <string>
#include <vector>
#include <windows.h>

using namespace std;

struct Discipline {
    string name;
    int mark;
};

struct Student {
    string fio;
    vector<Discipline> disc;
};

// Ввод данных студента
void readStudent(Student& s, const vector<string>& names, int num) {
    cout << endl << "Студент №" << num << endl;
    cout << "ФИО: ";
    getline(cin >> ws, s.fio);

    for (const string& nm : names) {
        Discipline d;
        d.name = nm;

        do {
            cout << "Балл по дисциплине " << nm << " (2-5): ";
            cin >> d.mark;

            if (d.mark < 2 || d.mark > 5) {
                cout << "Ошибка. Введите число от 2-5" << endl;
            }
        } while (d.mark < 2 || d.mark > 5);

        s.disc.push_back(d);
    }
}

// Вывод данных студента
void printStudent(const Student& s) {
    int sum = 0;

    cout << endl << s.fio << endl;

    for (const Discipline& d : s.disc) {
        cout << "  " << d.name << ": " << d.mark << endl;
        sum += d.mark;
    }

    cout << "  Среднее: " << fixed << setprecision(2)
         << (1.0 * sum / s.disc.size()) << endl;
}

// Вычисление среднего балла
double avg(const Student& s) {
    int sum = 0;

    for (const Discipline& d : s.disc) {
        sum += d.mark;
    }

    return 1.0 * sum / s.disc.size();
}

int main() {
    SetConsoleOutputCP(CP_UTF8);
    SetConsoleCP(CP_UTF8);

    int n, m;

    cout << "Количество студентов: ";
    cin >> n;

    if (n <= 0) {
        cout << "Число должно быть положительным." << endl;
        return 1;
    }

    cout << "Количество дисциплин: ";
    cin >> m;

    if (m < 4) {
        cout << "Ошибка.Введите не менее 4-х дисциплин" << endl;
        return 1;
    }

    vector<string> names(m);

    cout << endl << "Введите названия дисциплин" << endl;

    for (int i = 0; i < m; i++) {
        cout << "Дисциплина №" << i + 1 << ": ";
        getline(cin >> ws, names[i]);
    }

    vector<Student> st(n);

    for (int i = 0; i < n; i++) {
        readStudent(st[i], names, i + 1);
    }

    cout << endl << "  Все студенты и баллы по дисциплинам" << endl;

    for (const Student& s : st) {
        printStudent(s);
    }

    double mx = avg(st[0]);

    for (int i = 1; i < n; i++) {
        if (avg(st[i]) > mx) {
            mx = avg(st[i]);
        }
    }

    cout << endl << "Студенты с наибольшим средним баллом" << endl;
    cout << "   Наибольший средний балл: "
         << fixed << setprecision(2) << mx << endl;

    for (const Student& s : st) {
        if (avg(s) == mx) {
            printStudent(s);
        }
    }

    return 0;
}
```
